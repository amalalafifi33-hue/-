      const sc = DIALOGUE_SCENES[state.dialogueIndex];
      const audioEl = document.getElementById('dialogueAudioEl');
      const badge = document.getElementById('dialogueAudioBadge');
      if (!audioEl || !sc) return;

      clearTimeout(state.dialogueTimer);
      stopGeneratedNarration();
      audioEl.pause();

      if (sc.audio) {
        audioEl.src = sc.audio;
        audioEl.currentTime = 0;
        if (badge) badge.classList.remove('hidden');
        audioEl.onended = () => {
          if (badge) badge.classList.add('hidden');
          if (state.dialogueAutoPlay) {
            state.dialogueTimer = setTimeout(() => {
              if (!state.dialogueAutoPlay) return;
              if (state.dialogueIndex < DIALOGUE_SCENES.length - 1) {
                state.dialogueIndex++;
                renderDialogueScene(true);
              } else {
                state.dialogueAutoPlay = false;
                const autoBtn = document.getElementById('dlgAutoBtn');
                if (autoBtn) autoBtn.textContent = '▶ تشغيل تلقائي';
              }
            }, 550);
          }
        };
        audioEl.play().catch(() => {
          if (badge) badge.classList.add('hidden');
          speakDialogueFallback(sc.text);
        });
      } else {
        speakDialogueFallback(sc.text);
      }
    }

    function goToDialogueScene(idx) {
      state.dialogueIndex = idx;
      renderDialogueScene(true);
    }

    function openModal(id) {
      modalReturnFocus = document.activeElement;
      const modal = document.getElementById(id);
      modal.classList.remove('hidden');
      const heading = modal.querySelector('h2, h3');
      modal.setAttribute('role', 'dialog'); modal.setAttribute('aria-modal', 'true');
      if (heading) { heading.id ||= id + 'Title'; modal.setAttribute('aria-labelledby', heading.id); }
      improveSemantics(modal);
      requestAnimationFrame(() => modal.querySelector('button, input, select')?.focus());
    }
    function closeModal(id) {
      if(id==='passportModal')stopKasroonVoice();
      if (id === 'kasroonModal') { stopKasroonVoice(); document.getElementById('kasroonButton').setAttribute('aria-expanded','false'); }
      document.getElementById(id).classList.add('hidden');
      if (modalReturnFocus?.isConnected) modalReturnFocus.focus();
      if (id === 'dialogueModal') {
        state.dialogueAutoPlay = false;
        const autoBtn = document.getElementById('dlgAutoBtn');
        if (autoBtn) autoBtn.textContent = '▶ تشغيل تلقائي';
        stopDialogueAudio();
      }
    }

    function openDialogueModal() {
      openModal('dialogueModal');
      renderDialogueScene(!access.manualAudio);
    }

    function renderDialogueScene(autoSpeak = false) {
      const sc = DIALOGUE_SCENES[state.dialogueIndex];
      document.getElementById('sceneSpeakerLabel').textContent = `المتحدث: ${sc.speaker}`;
      document.getElementById('sceneCounter').textContent = `${state.dialogueIndex + 1} / ${DIALOGUE_SCENES.length}`;
      document.getElementById('dialogueBubble').innerHTML = `
        <p>${sc.text}</p>
        ${state.dialogueIndex === 2 ? renderPieSVG(1, 4, 60) : ''}
        ${state.dialogueIndex === 4 ? renderPieSVG(2, 4, 60, '#148080') : ''}
        ${state.dialogueIndex === 6 ? `<div class="flex gap-3 items-center">${renderPieSVG(2, 4, 54)} <span>=</span> ${renderPieSVG(1, 2, 54)}</div>` : ''}
        ${state.dialogueIndex === 7 ? renderPieSVG(4, 4, 60, '#1E7E5C') : ''}
        ${state.dialogueIndex === 9 ? `<div class="flex gap-2 items-center">${renderPieSVG(1, 4, 48)} <span>+</span> ${renderPieSVG(2, 4, 48, '#148080')} <span>=</span> ${renderPieSVG(3, 4, 48, '#1E7E5C')}</div>` : ''}
      `;
      const dotsEl = document.getElementById('dialogueSceneDots');
      if (dotsEl) {
        dotsEl.innerHTML = DIALOGUE_SCENES.map((item, i) => `
          <button onclick="goToDialogueScene(${i})" title="المشهد ${i + 1} (${item.speaker})"
            class="h-2.5 rounded-full transition-all ${
              i === state.dialogueIndex
                ? (item.speaker === 'ليان' ? 'w-6 bg-[#148080]' : 'w-6 bg-[#D96B43]')
                : 'w-2.5 bg-[#E6E1D6] hover:bg-[#9CA3AF]'
            }"></button>
        `).join('');
      }
      const layan = document.getElementById('avatarLayan');
      const zaid = document.getElementById('avatarZaid');
      if (sc.speaker === 'ليان') {
        layan.className = 'w-24 h-24 rounded-full object-cover border-4 border-[#148080] scale-105 shadow-md transition-all';
        zaid.className = 'w-24 h-24 rounded-full object-cover border-4 border-[#E6E1D6] opacity-65 transition-all';
      } else {
        zaid.className = 'w-24 h-24 rounded-full object-cover border-4 border-[#D96B43] scale-105 shadow-md transition-all';
        layan.className = 'w-24 h-24 rounded-full object-cover border-4 border-[#E6E1D6] opacity-65 transition-all';
      }
      const prevBtn = document.getElementById('dlgPrevBtn');
      const nextBtn = document.getElementById('dlgNextBtn');
      if (prevBtn) prevBtn.disabled = state.dialogueIndex === 0;
      if (nextBtn) nextBtn.disabled = state.dialogueIndex === DIALOGUE_SCENES.length - 1;

      if (autoSpeak) {
        playDialogueAudio();
      }
    }
