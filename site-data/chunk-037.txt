    function stopDialogueAudio() {
      clearTimeout(state.dialogueTimer);
      clearInterval(state.dialogueTimer);
      state.dialogueTimer = null;
      const audioEl = document.getElementById('dialogueAudioEl');
      if (audioEl) {
        audioEl.pause();
        audioEl.currentTime = 0;
        audioEl.onended = null;
      }
      const badge = document.getElementById('dialogueAudioBadge');
      if (badge) badge.classList.add('hidden');
      stopGeneratedNarration();
    }
