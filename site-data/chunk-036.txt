    function pickMatchRight(id) {
      playClickSound();
      state.selectedRight = id;
      checkMatchPair();
    }
    function pickMatchLeft(id) {
      playClickSound();
      state.selectedLeft = id;
      checkMatchPair();
    }
    function checkMatchPair() {
      if (state.selectedRight && state.selectedLeft) {
        if (state.selectedRight === state.selectedLeft) {
          state.matchedIds.add(state.selectedRight);
          playSuccessSound();
          showToast('إجابة صحيحة! أحسنت 👏');
          if (state.matchedIds.size === 4) {
            markTaskComplete(4, 'match');
            showToast('رائع! أكملت التوصيل وحصلت على نجمة ⭐');
          }
        } else {
          playErrorSound();
          showToast('حاول مرة أخرى، الأجزاء المظللة لا تطابق البسط');
        }
        state.selectedRight = null;
        state.selectedLeft = null;
      }
      renderStage();
      updateChrome();
    }
    function hintMatch() {
      const rem = MATCH_ITEMS.find(x => !state.matchedIds.has(x.id));
      if (rem) {
        state.selectedRight = rem.id;
        readKasroonHint(`تلميح: ابحث عن رسم يحتوي على ${rem.den} أجزاء متساوية، منها ${rem.num} مظلل.`);
        renderStage();
      }
    }
    function resetMatch() {
      state.matchedIds.clear();
      state.selectedLeft = null;
      state.selectedRight = null;
      renderStage();
    }
